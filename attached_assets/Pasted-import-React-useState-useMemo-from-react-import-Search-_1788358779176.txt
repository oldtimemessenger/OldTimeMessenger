/* ---------------------------------------------------------
   HELPERS
--------------------------------------------------------- */

function matchesQuery(node, q) {
  if (!q) return true;
  if (node.name.toLowerCase().includes(q)) return true;
  if (node.sub) return node.sub.some((c) => matchesQuery(c, q));
  return false;
}

function countSelected(node, selected) {
  let n = selected.has(node.id) ? 1 : 0;
  if (node.sub) n += node.sub.reduce((a, c) => a + countSelected(c, selected), 0);
  return n;
}

/* ---------------------------------------------------------
   NODE ROW (recursive)
--------------------------------------------------------- */

function Node({ node, depth, selected, toggle, expanded, toggleExpand, query }) {
  if (!matchesQuery(node, query)) return null;

  const hasSub = node.sub && node.sub.length > 0;
  const isSelected = selected.has(node.id);
  const isOpen = expanded.has(node.id) || (query && hasSub);
  const subCount = hasSub ? countSelected(node, selected) : 0;

  return (
    <div className={depth === 0 ? "" : ""}>
      <div
        className={[
          "flex items-center gap-3 transition-colors",
          depth === 0
            ? "px-4 py-4 rounded-2xl border"
            : "px-3 py-3 rounded-xl",
          depth === 0
            ? isSelected
              ? "bg-[#1B1B1F] border-[#1B1B1F]"
              : "bg-white border-[#E4E0D6]"
            : isSelected
            ? "bg-[#FBEAE7]"
            : "hover:bg-[#F1EFE8]",
        ].join(" ")}
        style={depth > 0 ? { marginLeft: depth * 16 } : undefined}
      >
        <button
          onClick={() => toggle(node.id)}
          className="flex-1 flex items-center justify-between text-left gap-3 min-w-0"
        >
          <div className="min-w-0">
            <div
              className={[
                depth === 0 ? "text-[15px] font-semibold" : "text-[14px] font-medium",
                depth === 0 ? (isSelected ? "text-[#F7F5F0]" : "text-[#1B1B1F]") : "text-[#1B1B1F]",
                "truncate",
              ].join(" ")}
            >
              {node.name}
            </div>
            {node.blurb && (
              <div className={depth === 0 ? "text-[13px] mt-0.5" : "text-xs mt-0.5"}
                style={{ color: depth === 0 ? (isSelected ? "#C9C6BD" : "#8A867C") : "#8A867C" }}>
                {node.blurb}
              </div>
            )}
            {hasSub && subCount > 0 && (
              <div className="text-xs mt-0.5" style={{ color: depth === 0 ? (isSelected ? "#E7A79C" : "#C23B2A") : "#C23B2A" }}>
                {subCount} selected
              </div>
            )}
          </div>

          <span
            className={[
              "shrink-0 w-6 h-6 rounded-full flex items-center justify-center border-2 transition-colors",
              isSelected
                ? depth === 0
                  ? "bg-[#D62828] border-[#D62828]"
                  : "bg-[#D62828] border-[#D62828]"
                : "border-[#D6D2C6] bg-transparent",
            ].join(" ")}
          >
            {isSelected && <Check size={14} strokeWidth={3} color="#FFF" />}
          </span>
        </button>

        {hasSub && (
          <button
            onClick={() => toggleExpand(node.id)}
            className={[
              "shrink-0 w-8 h-8 rounded-full flex items-center justify-center",
              depth === 0 ? (isSelected ? "text-[#F7F5F0]" : "text-[#1B1B1F]") : "text-[#1B1B1F]",
            ].join(" ")}
            aria-label="Expand"
          >
            <ChevronDown
              size={18}
              style={{ transform: isOpen ? "rotate(180deg)" : "rotate(0deg)", transition: "transform 160ms ease" }}
            />
          </button>
        )}
      </div>

      {hasSub && isOpen && (
        <div className="mt-1.5 mb-2 flex flex-col gap-1.5">
          {node.sub.map((child) => (
            <Node
              key={child.id}
              node={child}
              depth={depth + 1}
              selected={selected}
              toggle={toggle}
              expanded={expanded}
              toggleExpand={toggleExpand}
              query={query}
            />
          ))}
        </div>
      )}
    </div>
  );
}

/* ---------------------------------------------------------
   MAIN SCREEN
--------------------------------------------------------- */

export default function InterestsScreen() {
  const [query, setQuery] = useState("");
  const [selected, setSelected] = useState(new Set(["nba-los-angeles-lakers", "technology", "news-politics"]));
  const [expanded, setExpanded] = useState(new Set());
  const [langs, setLangs] = useState(new Set(["en"]));

  const toggle = (id) =>
    setSelected((prev) => {
      const next = new Set(prev);
      next.has(id) ? next.delete(id) : next.add(id);
      return next;
    });

  const toggleExpand = (id) =>
    setExpanded((prev) => {
      const next = new Set(prev);
      next.has(id) ? next.delete(id) : next.add(id);
      return next;
    });

  const toggleLang = (id) =>
    setLangs((prev) => {
      const next = new Set(prev);
      next.has(id) ? next.delete(id) : next.add(id);
      return next.size ? next : prev; // keep at least one
    });

  const q = query.trim().toLowerCase();
  const visible = useMemo(() => DATA.filter((n) => matchesQuery(n, q)), [q]);
  const totalSelected = selected.size;

  return (
    <div
      className="min-h-screen w-full flex flex-col"
      style={{ background: "#F7F5F0", fontFamily: "-apple-system, 'Inter', system-ui, sans-serif" }}
    >
      {/* Header */}
      <div className="sticky top-0 z-10" style={{ background: "#F7F5F0", borderBottom: "1px solid #E4E0D6" }}>
        <div className="px-5 pt-6 pb-4">
          <h1
            style={{
              fontFamily: "ui-serif, Georgia, 'Times New Roman', serif",
              fontSize: 28,
              fontWeight: 600,
              color: "#1B1B1F",
              letterSpacing: "-0.01em",
            }}
          >
            Your interests
          </h1>
          <p className="text-[14px] mt-1" style={{ color: "#6B6B65" }}>
            Choose topics and teams to shape what shows up in your feed.
          </p>
        </div>

        {/* Search */}
        <div className="px-5 pb-4">
          <div className="flex items-center gap-2 rounded-full px-4 py-2.5" style={{ background: "#FFFFFF", border: "1px solid #E4E0D6" }}>
            <Search size={16} color="#8A867C" />
            <input
              value={query}
              onChange={(e) => setQuery(e.target.value)}
              placeholder="Search topics, teams, leagues…"
              className="flex-1 bg-transparent outline-none text-[14px]"
              style={{ color: "#1B1B1F" }}
            />
            {query && (
              <button onClick={() => setQuery("")} aria-label="Clear search">
                <X size={15} color="#8A867C" />
              </button>
            )}
          </div>
        </div>

        {/* Language pills */}
        <div className="px-5 pb-4 overflow-x-auto">
          <div className="text-xs font-medium mb-2" style={{ color: "#8A867C" }}>
            CONTENT LANGUAGES
          </div>
          <div className="flex gap-2 flex-wrap">
            {LANGUAGES.map((l) => {
              const active = langs.has(l.id);
              return (
                <button
                  key={l.id}
                  onClick={() => toggleLang(l.id)}
                  className="px-3.5 py-2 rounded-full text-[13px] font-medium transition-colors whitespace-nowrap"
                  style={{
                    background: active ? "#1B1B1F" : "#FFFFFF",
                    color: active ? "#F7F5F0" : "#1B1B1F",
                    border: `1px solid ${active ? "#1B1B1F" : "#E4E0D6"}`,
                  }}
                >
                  {l.native}
                </button>
              );
            })}
          </div>
        </div>
      </div>

      {/* List */}
      <div className="flex-1 px-5 py-4 flex flex-col gap-3 pb-28">
        {visible.length === 0 && (
          <div className="text-center py-16" style={{ color: "#8A867C" }}>
            No topics match "{query}"
          </div>
        )}
        {visible.map((node) => (
          <Node
            key={node.id}
            node={node}
            depth={0}
            selected={selected}
            toggle={toggle}
            expanded={expanded}
            toggleExpand={toggleExpand}
            query={q}
          />
        ))}
      </div>

      {/* Sticky footer */}
      <div
        className="fixed bottom-0 left-0 right-0 px-5 py-4"
        style={{ background: "#F7F5F0", borderTop: "1px solid #E4E0D6" }}
      >
        <button
          className="w-full rounded-full py-3.5 text-[15px] font-semibold flex items-center justify-center gap-2"
          style={{ background: "#D62828", color: "#FFFFFF" }}
        >
          Done — {totalSelected} selected
        </button>
      </div>
    </div>
  );
}
